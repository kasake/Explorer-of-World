        function redrawFog() {

            if (!fogCtx || !map) return;

            const width = fogCanvas.width;

            const height = fogCanvas.height;

 

            // 1. Fill entire canvas with dark fog using configured opacity

            fogCtx.clearRect(0, 0, width, height);

            fogCtx.fillStyle = `rgba(7, 10, 19, ${fogOpacity})`;

            fogCtx.fillRect(0, 0, width, height);

 

            // 2. Cut out explored areas using destination-out

            fogCtx.globalCompositeOperation = 'destination-out';

 

            const zoom = map.getZoom();

 

            exploredPoints.forEach(pt => {

                const latLng = L.latLng(pt.lat, pt.lng);

                const layerPoint = map.latLngToLayerPoint(latLng);

                const containerPoint = map.layerPointToContainerPoint(layerPoint);

 

                const res = 156543.03392 * Math.cos(latLng.lat * Math.PI / 180) / Math.pow(2, zoom);

                const radiusPx = pt.radius / res;

 

                fogCtx.beginPath();

                fogCtx.arc(containerPoint.x, containerPoint.y, Math.max(radiusPx, 8), 0, Math.PI * 2);

                fogCtx.fill();

            });

 

            fogCtx.globalCompositeOperation = 'source-over';

        }

        // --- GPS Handling ---

        function startGPSWatch() {

            if (!navigator.geolocation) {

                showToast('您的浏览器不支持 Geolocation API', 'error');

                return;

            }

 

            watchId = navigator.geolocation.watchPosition(

                (position) => {

                    const lat = position.coords.latitude;

                    const lng = position.coords.longitude;

                    const speedMs = position.coords.speed || 0;

                    currentAccuracy = position.coords.accuracy;

                    currentSpeed = speedMs * 3.6; // convert to km/h

                    if (currentSpeed > maxSpeed) maxSpeed = currentSpeed;

 

                    // Accuracy Filter: discard data if accuracy error is larger than set threshold

                    if (maxAccuracyThreshold > 0 && currentAccuracy > maxAccuracyThreshold) {

                        updateUserPosition({ lat, lng }, false, currentAccuracy);

                        updateStatsUI();

                        showToast(`信号微弱 (精度 ±${Math.round(currentAccuracy)}m > 过滤门限 ${maxAccuracyThreshold}m)，已丢弃该数据点`, 'info');

                        return;

                    }

 

                    const newPos = { lat, lng };

                    const dist = calculateDistance(currentPos.lat, currentPos.lng, lat, lng);

 

                    if (dist > 3) { // Minimum 3 meters movement to record

                        totalDistance += dist / 1000;

                        currentPos = newPos;

                        updateUserPosition(currentPos, true, currentAccuracy);

                        addExploredPoint(currentPos.lat, currentPos.lng, FOG_RADIUS_METERS);

                        updateStatsUI();

                        saveToLocalStorage();

                    } else {

                        updateUserPosition(currentPos, false, currentAccuracy);

                        updateStatsUI();

                    }

                },

                (error) => {

                    let errMsg = '获取GPS信号出错';

                    if (error) {

                        switch (error.code) {

                            case 1:

                                errMsg = 'GPS权限被拒绝，请允许位置权限以进行探索';

                                break;

                            case 2:

                                errMsg = '无法获取当前位置信息';

                                break;

                            case 3:

                                errMsg = '获取GPS位置超时';

                                break;

                            default:

                                errMsg = error.message || '未知GPS错误';

                        }

                    }

                    console.warn('GPS watch error code:', error.code, 'message:', error.message);

                    showToast(errMsg, 'error');

                },

                { enableHighAccuracy: true, maximumAge: 0, timeout: 10000 }

            );

        }

        // --- Position & Map Utilities ---

        function updateUserPosition(pos, panMap = false, accuracy = null) {

            if (userMarker) {

                userMarker.setLatLng([pos.lat, pos.lng]);

            }

            if (accuracyCircle) {

                if (accuracy !== null && accuracy !== undefined) {

                    accuracyCircle.setLatLng([pos.lat, pos.lng]);

                    accuracyCircle.setRadius(accuracy);

                }

            }

            if (panMap && map) {

                map.panTo([pos.lat, pos.lng]);

            }

 

            if (isTracking) {

                trackHistory.push({ lat: pos.lat, lng: pos.lng, time: Date.now() });

                const latLngs = trackHistory.map(p => [p.lat, p.lng]);

                trackPolyline.setLatLngs(latLngs);

            }

        }

        function calculateDistance(lat1, lon1, lat2, lon2) {

            const R = 6371e3; // Earth radius in meters

            const φ1 = lat1 * Math.PI / 180;

            const φ2 = lat2 * Math.PI / 180;

            const Δφ = (lat2 - lat1) * Math.PI / 180;

            const Δλ = (lon2 - lon1) * Math.PI / 180;

 

            const a = Math.sin(Δφ / 2) * Math.sin(Δφ / 2) +

                      Math.cos(φ1) * Math.cos(φ2) *

                      Math.sin(Δλ / 2) * Math.sin(Δλ / 2);

            const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));

 

            return R * c; // meters

        }

        function saveToLocalStorage() {

            const data = {

                exploredPoints,

                trackHistory,

                totalDistance,

                elapsedSeconds,

        function loadFromLocalStorage() {

            const saved = localStorage.getItem('fog_of_world_data');

            if (saved) {

                try {

                    const data = JSON.parse(saved);

                    exploredPoints = data.exploredPoints || [];

                    trackHistory = data.trackHistory || [];

                    totalDistance = data.totalDistance || 0;

                    elapsedSeconds = data.elapsedSeconds || 0;

                    if (data.currentPos) currentPos = data.currentPos;

                    if (data.fogOpacity !== undefined) fogOpacity = data.fogOpacity;

                    if (data.maxAccuracyThreshold !== undefined) maxAccuracyThreshold = data.maxAccuracyThreshold;

                    if (data.themeMode) themeMode = data.themeMode;

 

                    // Sync UI for settings

                    const opacitySlider = document.getElementById('fogOpacitySlider');

                    if (opacitySlider) opacitySlider.value = fogOpacity;

                    const opacityVal = document.getElementById('fogOpacityVal');

                    if (opacityVal) opacityVal.innerText = Math.round(fogOpacity * 100) + '%';

 

                    const filterSlider = document.getElementById('accuracyFilterSlider');

                    if (filterSlider) filterSlider.value = maxAccuracyThreshold;

                    const valElem = document.getElementById('accuracyFilterVal');

                    if (valElem) {

                        valElem.innerText = maxAccuracyThreshold === 0 ? '关闭过滤' : `≤ ${maxAccuracyThreshold} 米`;

                    }

 

                    if (userMarker) userMarker.setLatLng([currentPos.lat, currentPos.lng]);

                    if (trackPolyline) {

                        const latLngs = trackHistory.map(p => [p.lat, p.lng]);

                        trackPolyline.setLatLngs(latLngs);

                    }

                    redrawFog();

                } catch (e) {

                    console.error('Failed to parse local storage fog data:', e);

                }

            }

        }

        function importData(event) {

            const file = event.target.files[0];

            if (!file) return;

 

            const reader = new FileReader();

            reader.onload = function(e) {

                try {

                    const imported = JSON.parse(e.target.result);

                    if (imported.exploredPoints) exploredPoints = imported.exploredPoints;

                    if (imported.trackHistory) trackHistory = imported.trackHistory;

                    if (imported.totalDistance !== undefined) totalDistance = imported.totalDistance;

                    if (imported.elapsedSeconds !== undefined) elapsedSeconds = imported.elapsedSeconds;

 

                    saveToLocalStorage();

                    redrawFog();

                    updateStatsUI();

                    closeBackupSubModal();

                    closeSettingsModal();

                    showToast('数据导入成功！', 'success');

                } catch (err) {

                    showToast('导入失败，无效的 JSON 文件格式', 'error');

                }

            };

            reader.readAsText(file);

        }

        // --- Toast Notifications & Modal Controls ---

        function showToast(message, type = 'info') {

            const container = document.getElementById('toastContainer');

            if (!container) return;

 

            const toast = document.createElement('div');

            let bgStyle = 'bg-slate-900 border-slate-700 text-slate-200';

            let iconClass = 'fa-solid fa-circle-info text-cyan-400';

 

            if (type === 'success') {

                bgStyle = 'bg-slate-900 border-emerald-500/50 text-emerald-200';

                iconClass = 'fa-solid fa-circle-check text-emerald-400';

            } else if (type === 'error') {

                bgStyle = 'bg-slate-900 border-rose-500/50 text-rose-200';

                iconClass = 'fa-solid fa-circle-exclamation text-rose-400';

            }

 

            toast.className = `flex items-center gap-2.5 px-4 py-2.5 rounded-2xl border shadow-xl text-xs backdrop-blur-md transition-all duration-300 transform opacity-0 translate-y-[-10px] pointer-events-auto ${bgStyle}`;

            toast.innerHTML = `<i class="${iconClass}"></i><span>${message}</span>`;

 

            container.appendChild(toast);

 

            // Trigger enter animation

            requestAnimationFrame(() => {

                toast.classList.remove('opacity-0', 'translate-y-[-10px]');

            });

 

            // Dismiss after 3 seconds

            setTimeout(() => {

                toast.classList.add('opacity-0', 'translate-y-[-10px]');

                setTimeout(() => toast.remove(), 300);

            }, 3000);

        }
